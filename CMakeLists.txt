cmake_minimum_required(VERSION 3.20)
project(PBC_gAPI VERSION 0.2.0 LANGUAGES C CXX)
option(PBC_BUILD_EXAMPLES "Build standalone example programs" ON)
option(PBC_BUILD_TESTS "Build regression tests" ON)
option(PBC_ENABLE_GLES "Build the portable OpenGL ES 3 backend and EGL helper" OFF)
option(PBC_ENABLE_CUDA "Build the NVIDIA CUDA backend" OFF)
option(PBC_ENABLE_SDL "Build SDL2 presentation (Raspbian/Linux)" OFF)
option(PBC_ENABLE_OPENCL "Build custom compute rasterizer for native Intel OpenCL GPUs" OFF)
option(PBC_OPENCL_TEST_DEVICE "Require functional OpenCL tests on an explicitly selected test device (e.g. PoCL CPU)" OFF)
option(PBC_SANITIZERS "Enable address/undefined sanitizers on GCC/Clang" OFF)
find_package(Threads REQUIRED)
add_library(pbc STATIC src/framebuffer.cpp src/pipeline.cpp src/cpu_renderer.cpp src/obj.cpp)
add_library(pbc::pbc ALIAS pbc)
target_include_directories(pbc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(pbc PUBLIC cxx_std_17)
target_link_libraries(pbc PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(pbc PRIVATE /W4 /permissive-)
else()
  target_compile_options(pbc PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PBC_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pbc PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pbc PUBLIC -fsanitize=address,undefined)
endif()
if(WIN32)
  target_sources(pbc PRIVATE src/surface_hdc.cpp)
  target_link_libraries(pbc PRIVATE gdi32 user32)
endif()
if(ANDROID)
  target_sources(pbc PRIVATE src/surface_android.cpp)
  target_link_libraries(pbc PUBLIC android log)
endif()
if(PBC_ENABLE_GLES)
  if(ANDROID)
    find_library(PBC_EGL EGL REQUIRED)
    find_library(PBC_GLES GLESv3 REQUIRED)
  else()
    find_path(PBC_GLES_INCLUDE GLES3/gl3.h REQUIRED)
    find_path(PBC_EGL_INCLUDE EGL/egl.h REQUIRED)
    find_library(PBC_EGL EGL REQUIRED)
    find_library(PBC_GLES GLESv2 REQUIRED)
    target_include_directories(pbc PRIVATE ${PBC_GLES_INCLUDE} ${PBC_EGL_INCLUDE})
  endif()
  target_sources(pbc PRIVATE src/gles_renderer.cpp src/egl_context.cpp)
  target_link_libraries(pbc PUBLIC ${PBC_EGL} ${PBC_GLES})
  target_compile_definitions(pbc PUBLIC PBC_HAS_GLES=1)
endif()
if(PBC_ENABLE_SDL)
  find_package(SDL2 REQUIRED CONFIG)
  target_sources(pbc PRIVATE src/surface_sdl.cpp)
  target_link_libraries(pbc PUBLIC SDL2::SDL2)
  target_compile_definitions(pbc PUBLIC PBC_HAS_SDL=1 SDL_MAIN_HANDLED=1)
endif()
if(PBC_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit 12.6 REQUIRED)
  add_library(pbc_cuda STATIC src/cuda_renderer.cu)
  target_link_libraries(pbc_cuda PUBLIC pbc CUDA::cudart)
  set_target_properties(pbc_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
    CUDA_SEPARABLE_COMPILATION ON CUDA_RESOLVE_DEVICE_SYMBOLS OFF)
  target_compile_definitions(pbc_cuda PUBLIC PBC_HAS_CUDA=1)
  find_path(PBC_NVML_INCLUDE nvml.h HINTS ${CUDAToolkit_INCLUDE_DIRS})
  if(PBC_NVML_INCLUDE)
    target_include_directories(pbc_cuda PRIVATE ${PBC_NVML_INCLUDE})
    target_compile_definitions(pbc_cuda PRIVATE PBC_HAS_NVML=1)
    target_link_libraries(pbc_cuda PRIVATE ${CMAKE_DL_LIBS})
  endif()
endif()
if(PBC_ENABLE_OPENCL)
  find_package(OpenCL 1.2 REQUIRED)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/opencl_kernel.cl")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/opencl_kernel.cl" PBC_OPENCL_KERNEL)
  configure_file(src/opencl_source.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/pbc_opencl_source.hpp" @ONLY)
  add_library(pbc_opencl STATIC src/opencl_renderer.cpp)
  target_include_directories(pbc_opencl PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(pbc_opencl PUBLIC PBC_HAS_OPENCL=1 PRIVATE CL_TARGET_OPENCL_VERSION=120)
  target_link_libraries(pbc_opencl PUBLIC pbc OpenCL::OpenCL)
  if(MSVC)
    target_compile_options(pbc_opencl PRIVATE /W4 /permissive-)
  else()
    target_compile_options(pbc_opencl PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
if(PBC_BUILD_EXAMPLES AND NOT ANDROID)
  add_executable(pbc_cpu examples/cpu.cpp)
  target_link_libraries(pbc_cpu PRIVATE pbc)
  add_executable(pbc_benchmark examples/benchmark.cpp)
  target_link_libraries(pbc_benchmark PRIVATE pbc)
  if(PBC_ENABLE_OPENCL)
    add_executable(pbc_opencl_example examples/opencl.cpp)
    target_link_libraries(pbc_opencl_example PRIVATE pbc_opencl)
  endif()
  if(PBC_ENABLE_GLES)
    add_executable(pbc_gles examples/gles.cpp)
    target_link_libraries(pbc_gles PRIVATE pbc)
  endif()
  if(PBC_ENABLE_CUDA)
    add_executable(pbc_cuda_example examples/cuda.cpp)
    target_link_libraries(pbc_cuda_example PRIVATE pbc_cuda)
  endif()
  if(WIN32 OR PBC_ENABLE_SDL)
    add_executable(pbc_window examples/window.cpp)
    target_link_libraries(pbc_window PRIVATE pbc)
  endif()
endif()
if(ANDROID AND PBC_BUILD_EXAMPLES)
  add_library(pbc_native SHARED examples/android/native.cpp ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
  target_include_directories(pbc_native PRIVATE ${ANDROID_NDK}/sources/android/native_app_glue)
  target_link_libraries(pbc_native PRIVATE pbc)
  set_target_properties(pbc PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_options(pbc_native PRIVATE -uANativeActivity_onCreate)
endif()
if(PBC_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(pbc_tests tests/core.cpp)
  target_link_libraries(pbc_tests PRIVATE pbc)
  target_compile_definitions(pbc_tests PRIVATE PBC_ASSET_PATH="${CMAKE_CURRENT_SOURCE_DIR}/PBC_GAPI_CUDA/suzanne.obj")
  foreach(case texture obj raster clipping interpolation parallel shaders mrt validation asset)
    add_test(NAME pbc_${case} COMMAND pbc_tests ${case})
  endforeach()
  if(PBC_ENABLE_OPENCL)
    add_executable(pbc_opencl_tests tests/opencl.cpp)
    target_link_libraries(pbc_opencl_tests PRIVATE pbc_opencl)
    add_test(NAME pbc_intel_opencl COMMAND pbc_opencl_tests --intel-gpu)
    set_tests_properties(pbc_intel_opencl PROPERTIES SKIP_RETURN_CODE 77)
    if(PBC_OPENCL_TEST_DEVICE)
      add_test(NAME pbc_opencl_compute COMMAND pbc_opencl_tests --require-testing-device)
    endif()
  endif()
  if(PBC_ENABLE_GLES)
    add_executable(pbc_gles_tests tests/gles.cpp)
    target_link_libraries(pbc_gles_tests PRIVATE pbc)
    target_include_directories(pbc_gles_tests PRIVATE ${PBC_GLES_INCLUDE} ${PBC_EGL_INCLUDE})
    add_test(NAME pbc_gles COMMAND pbc_gles_tests)
    set_tests_properties(pbc_gles PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  if(PBC_ENABLE_CUDA)
    add_executable(pbc_cuda_tests tests/cuda.cu)
    target_link_libraries(pbc_cuda_tests PRIVATE pbc_cuda)
    set_target_properties(pbc_cuda_tests PROPERTIES CUDA_STANDARD 17 CUDA_SEPARABLE_COMPILATION ON)
    add_test(NAME pbc_cuda COMMAND pbc_cuda_tests)
    set_tests_properties(pbc_cuda PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
